import yfinance as yf
from pykalman import KalmanFilter
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Download data for two stocks
data = yf.download(["EWC", "EWA"], start="2017-01-01", end="2022-12-31")

# Calculate the ratio of the closing prices for pairs trading
data['Ratio'] = data['Close']['EWA'] / data['Close']['EWC']

# Initialize Kalman Filter
kf = KalmanFilter(
    transition_matrices=[1],
    observation_matrices=[1],
    initial_state_mean=0,
    initial_state_covariance=1,
    observation_covariance=1,
    transition_covariance=.0001
)

# Apply Kalman Filter to the ratio data
state_means, state_covs = kf.filter(data['Ratio'])

# Extract mean and covariance from the Kalman Filter results
data['mean'] = state_means.squeeze()
data['cov'] = state_covs.squeeze()
data['std'] = np.sqrt(data['cov'])

# Drop NaN values from DataFrame
data.dropna(inplace=True)

# Calculate rolling mean and z_score
data['ma'] = data['Ratio'].rolling(5).mean()
data['z_score'] = (data['ma'] - data['mean']) / data['std']

# Plotting
plt.figure(figsize=(15, 7))
plt.plot(data.index, data['Ratio'] - data['mean'], 'm', lw=1)  # Plotting error: real_value - mean
plt.plot(data.index, np.sqrt(data['cov']), 'y', lw=1)  # Plotting standard deviation
plt.plot(data.index, -np.sqrt(data['cov']), 'c', lw=1)  # Plotting negative standard deviation
plt.title('Kalman filter estimate')
plt.legend(['Error: real_value - mean', 'std', '-std'])
plt.xlabel('Date')
plt.ylabel('Value')
plt.show()
