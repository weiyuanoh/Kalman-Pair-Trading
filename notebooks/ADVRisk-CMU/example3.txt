# ---------------------------------------------------------
# 2. Long-run Engle–Granger regression
# ---------------------------------------------------------
result = smf.ols("MSFT ~ AAPL", data=df).fit()
print("\n==============================")
print("Long-Run Cointegration Regression (Engle–Granger Step 1)")
print("==============================")
print(result.summary())

# Extract residuals
df["resid"] = result.resid

# ---------------------------------------------------------
# 3. ADF test on residuals (Engle–Granger Step 2)
# ---------------------------------------------------------
print("\n==============================")
print("ADF Test on Residuals (Cointegration Test)")
print("==============================")

adf_stat, pval, lags, nobs, crit, icbest = adfuller(df["resid"].dropna(), regression="c")

print(f"ADF Statistic: {adf_stat}")
print(f"p-value: {pval}")
print(f"Lags used: {lags}")
print(f"Observations: {nobs}")
print("Critical values:")
for k, v in crit.items():
    print(f"   {k}: {v}")



# ---------------------------------------------------------
# 4. Build the Error-Correction Model (ECM)
# ---------------------------------------------------------
df["dMSFT"] = df["MSFT"].diff()
df["dAAPL"] = df["AAPL"].diff()
df["ECT_lag"] = df["resid"].shift(1)

# Add short-run lags
df["dMSFT_lag1"] = df["dMSFT"].shift(1)
df["dAAPL_lag1"] = df["dAAPL"].shift(1)

ecm_df = df.dropna()

# ECM regression:
ecm_model = smf.ols(
    "dMSFT ~ ECT_lag + dMSFT_lag1 + dAAPL_lag1",
    data=ecm_df
).fit()

print("\n==============================")
print("Error-Correction Model (ECM) Implied by Engle–Granger")
print("==============================")
print(ecm_model.summary())



# ---------------------------------------------------------
# 5. Print the ECM equation with estimated coefficients
# ---------------------------------------------------------
alpha = ecm_model.params["ECT_lag"]
gamma1 = ecm_model.params["dMSFT_lag1"]
delta1 = ecm_model.params["dAAPL_lag1"]
const = ecm_model.params["Intercept"]

print("\n==============================")
print("Estimated ECM Equation (Explicit Form)")
print("==============================")

print(f"ΔMSFT_t = {const:.6f}"
